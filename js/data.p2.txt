  const TRAIL_MAX = 300;
  const TRAIL_MIN_METERS = 15;
  const TRAIL_MIN_MS = 45000;

  function haversineMeters(lat1, lng1, lat2, lng2) {
    const R = 6371000;
    const toRad = function (d) {
      return (d * Math.PI) / 180;
    };
    const dLat = toRad(lat2 - lat1);
    const dLng = toRad(lng2 - lng1);
    const a =
      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
      Math.cos(toRad(lat1)) *
        Math.cos(toRad(lat2)) *
        Math.sin(dLng / 2) *
        Math.sin(dLng / 2);
    return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  }

  function shouldAppendTrail(prevTrail, lat, lng, atIso) {
    if (!prevTrail || !prevTrail.length) return true;
    const last = prevTrail[prevTrail.length - 1];
    if (!last || last.lat == null || last.lng == null) return true;
    const dist = haversineMeters(last.lat, last.lng, lat, lng);
    if (dist >= TRAIL_MIN_METERS) return true;
    const t0 = last.at ? Date.parse(last.at) : 0;
    const t1 = atIso ? Date.parse(atIso) : Date.now();
    if (t1 - t0 >= TRAIL_MIN_MS) return true;
    return false;
  }

  function appendTrailPoint(prevTrail, lat, lng, atIso) {
    const trail = Array.isArray(prevTrail) ? prevTrail.slice() : [];
    if (!shouldAppendTrail(trail, lat, lng, atIso)) return trail;
    trail.push({ lat: Number(lat), lng: Number(lng), at: atIso || nowIso() });
    if (trail.length > TRAIL_MAX) return trail.slice(trail.length - TRAIL_MAX);
    return trail;
  }

  async function updateTeamGps(projectId, teamId, coords) {
    if (!coords || coords.lat == null || coords.lng == null) return null;

      data.teamSessions[projectId][teamId] = Object.assign({}, prev, {
        lastLat: lat,
        lastLng: lng,
        lastGpsAt: at,
        trail: trail
      });
      saveDemo(data);
      return data.teamSessions[projectId][teamId];
    }
    const ref = db
      .collection("projects")
      .doc(projectId)
      .collection("teamSessions")
      .doc(teamId);
    const doc = await ref.get();
    const prev = doc.exists ? doc.data() || {} : {};
    const trail = appendTrailPoint(prev.trail, lat, lng, at);
    await ref.set(
      {
        lastLat: lat,
        lastLng: lng,
        lastGpsAt: at,
        trail: trail
      },
      { merge: true }
    );
    return getTeamSession(projectId, teamId);
  }

  /**
   * Clear runtime data for a project: submissions + team progress/GPS trail.
   * Keeps project settings, locations/questions, and team accounts.
   */
  async function clearProjectRuntimeData(projectId) {
    if (demo) {
      const data = loadDemo();
      data.submissions[projectId] = {};
      const sessions = data.teamSessions[projectId] || {};
      Object.keys(sessions).forEach(function (tid) {
        const prev = sessions[tid] || {};
        sessions[tid] = {
          lastLoginAt: prev.lastLoginAt || null,
          completedLocationIds: [],
          trail: [],
          unlockedEdits: {},
          locationAnswers: {}
        };
      });
      data.teamSessions[projectId] = sessions;
      saveDemo(data);
      return { submissionsCleared: true, sessionsReset: Object.keys(sessions).length };
    }
    const projRef = db.collection("projects").doc(projectId);
    const subSnap = await projRef.collection("submissions").get();
    const sessSnap = await projRef.collection("teamSessions").get();
    let batch = db.batch();
    let ops = 0;
    const commitIfNeeded = async function () {
      if (ops >= 400) {
        await batch.commit();
        batch = db.batch();
        ops = 0;
      }
    };
    for (let i = 0; i < subSnap.docs.length; i++) {
      batch.delete(subSnap.docs[i].ref);
      ops++;
      await commitIfNeeded();
    }
    for (let i = 0; i < sessSnap.docs.length; i++) {
      const d = sessSnap.docs[i];
      const prev = d.data() || {};
      batch.set(d.ref, {
        lastLoginAt: prev.lastLoginAt || null,
        completedLocationIds: [],
        trail: [],
        unlockedEdits: {},
        locationAnswers: {}
      });
      ops++;
      await commitIfNeeded();
    }
    if (ops > 0) await batch.commit();
    return {
      submissionsCleared: subSnap.size,
      sessionsReset: sessSnap.size
    };
  }

  /**
   * Rewrite location order to 1..n contiguous and persist changed docs.
   * orderedIds: array of location ids in desired order.
   */
  async function reorderLocations(projectId, orderedIds) {
    const locs = await listLocations(projectId);
    const byId = {};
    locs.forEach((l) => {
      byId[l.id] = l;
    });
    const ids = (orderedIds || []).filter((id) => byId[id]);
    locs.forEach((l) => {
      if (ids.indexOf(l.id) < 0) ids.push(l.id);
    });
    const updates = [];
    ids.forEach((id, i) => {
      const nextOrder = i + 1;
      if ((byId[id].order || 0) !== nextOrder) {
        updates.push({ id, order: nextOrder, loc: byId[id] });
      }
    });
    if (!updates.length) return listLocations(projectId);
    if (demo) {
      const data = loadDemo();
      if (!data.locations[projectId]) data.locations[projectId] = {};
      updates.forEach((u) => {
        const cur = data.locations[projectId][u.id] || {};
        data.locations[projectId][u.id] = Object.assign({}, cur, { order: u.order });
      });
      if (data.projects[projectId]) data.projects[projectId].updatedAt = nowIso();
      saveDemo(data);
      return listLocations(projectId);
    }
    const batch = db.batch();
    const col = db.collection("projects").doc(projectId).collection("locations");
    updates.forEach((u) => {
      batch.set(col.doc(u.id), { order: u.order }, { merge: true });
    });
    await batch.commit();
    try {
      await db.collection("projects").doc(projectId).update({ updatedAt: nowIso() });
    } catch (e) {
      console.warn("reorderLocations: updatedAt touch failed", e && e.message);
    }
    return listLocations(projectId);
  }

  function exportSubmissionsCSV(project, locations, submissions) {
    const headers = ["提交時間", "隊伍", "地區", "緯度", "經度"];
    locations.forEach((l) => {
      (l.questions || []).forEach((q) => {
        headers.push(l.name + " - " + q.text);
      });
    });
    const rows = [headers.map((h) => '"' + String(h).replace(/"/g, '""') + '"').join(",")];
    submissions.forEach((s) => {
      const loc = locations.find((l) => l.id === s.locationId);
      const row = [
        s.submittedAt ? new Date(s.submittedAt).toLocaleString("zh-HK") : "",
        s.teamName || "",
        loc ? loc.name : s.locationId,
        s.coords ? s.coords.lat : "",
        s.coords ? s.coords.lng : ""
      ];
      locations.forEach((l) => {
        (l.questions || []).forEach((q) => {
          if (l.id === s.locationId) {
            row.push((s.answers && s.answers[q.id]) || "");
          } else {
            row.push("");
          }
        });
      });
      rows.push(row.map((c) => '"' + String(c).replace(/"/g, '""') + '"').join(","));
    });
    return "\uFEFF" + rows.join("\n");
  }

        const reader = new FileReader();
        reader.onload = () => resolve(reader.result);
        reader.onerror = reject;
        reader.readAsDataURL(file);
      });
    }
    if (!storage) throw new Error("Storage 未啟用");
    const path =
      "projects/" + projectId + "/images/" + Date.now() + "_" + (file.name || "img");
    const ref = storage.ref().child(path);
    await ref.put(file);
    return await ref.getDownloadURL();
  }
