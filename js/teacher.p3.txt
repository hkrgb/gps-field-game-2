
  let monitorProjectId = null;
  let monitorCache = null;

  function nearestCheckpointInfo(lat, lng, locs) {
    if (lat == null || lng == null || !locs || !locs.length) return null;
    let best = null;
    locs.forEach(function (loc) {
      if (loc.lat == null || loc.lng == null) return;
      const dist = GPS.calculateDistance(lat, lng, loc.lat, loc.lng);
      if (!best || dist < best.dist) {
        best = { loc: loc, dist: dist };
      }
    });
    if (!best) return null;
    const bearing = GPS.bearingDegrees(best.loc.lat, best.loc.lng, lat, lng);
    const dir = GPS.compassLabel(bearing);
    return {
      name: best.loc.name,
      dist: best.dist,
      bearing: bearing,
      dir: dir,
      phrase:
        "約在「" +
        best.loc.name +
        "」的" +
        dir +
        "方向 " +
        GPS.formatDistance(best.dist)
    };
  }

  function buildTeamProgressHtml(t, s, locs) {
    const doneIds = new Set(s.completedLocationIds || []);
    const done = doneIds.size;
    const total = locs.length;
    const pct = total ? Math.round((done / total) * 100) : 0;
    let checks = "";
    locs.forEach(function (loc) {
      const ok = doneIds.has(loc.id);
      checks +=
        '<span class="inline-flex items-center gap-0.5 text-[10px] px-1.5 py-0.5 rounded ' +
        (ok ? "bg-emerald-100 text-emerald-800" : "bg-slate-100 text-slate-500") +
        '" title="' +
        esc(loc.name) +
        '">' +
        (ok ? "✓" : "○") +
        " " +
        esc(loc.name) +
        "</span>";
    });
    let gpsLine = '<span class="text-slate-400">尚未回報位置</span>';
    if (s.lastLat != null && s.lastLng != null) {
      const info = nearestCheckpointInfo(s.lastLat, s.lastLng, locs);
      if (info) {
        gpsLine =
          '<span class="text-emerald-800"><i class="fa-solid fa-location-crosshairs mr-1"></i>' +
          esc(info.phrase) +
          "</span>";
      } else {
        gpsLine =
          '<span class="text-slate-600">位置 ' +
          Number(s.lastLat).toFixed(5) +
          ", " +
          Number(s.lastLng).toFixed(5) +
          "</span>";
      }
      if (s.lastGpsAt) {
        gpsLine +=
          '<span class="text-slate-400 ml-1">· ' +
          new Date(s.lastGpsAt).toLocaleString("zh-HK") +
          "</span>";
      }
    }
    return (
      '<div class="font-bold text-slate-800 flex justify-between items-center gap-2">' +
      "<span>" +
      esc(t.name) +
      '</span><span class="text-xs font-semibold text-emerald-700">完成 ' +
      done +
      "/" +
      total +
      "</span></div>" +
      '<div class="mt-1.5 h-1.5 bg-slate-100 rounded-full overflow-hidden">' +
      '<div class="h-full bg-emerald-500 rounded-full" style="width:' +
      pct +
      '%"></div></div>' +
      '<div class="mt-2 flex flex-wrap gap-1">' +
      (checks || '<span class="text-[10px] text-slate-400">未設定地區</span>') +
      "</div>" +
      '<div class="mt-2 text-[11px] leading-snug">' +
      gpsLine +
      "</div>" +
      '<div class="text-[10px] text-slate-400 mt-1">上次登入：' +
      (s.lastLoginAt
        ? new Date(s.lastLoginAt).toLocaleString("zh-HK")
        : "尚未登入") +
      " · 點擊查看答題</div>"
    );
  }

  function renderTeamDetail(teamId) {
    const panel = document.getElementById("monTeamDetail");
    if (!panel || !monitorCache) return;
    const t =
      monitorCache.teams.find(function (x) {
        return x.id === teamId;
      }) || { id: teamId, name: teamId };
    const locs = monitorCache.locations;
    const subs = (monitorCache.submissions || []).filter(function (s) {
      return s.teamId === teamId;
    });
    let html =
      '<div class="flex justify-between items-center gap-2 mb-2">' +
      '<h4 class="font-bold text-sm text-slate-800">答題詳情：' +
      esc(t.name) +
      "</h4>" +
      '<button type="button" id="btnCloseMonDetail" class="text-xs text-slate-500 hover:underline">關閉</button></div>';
    if (!subs.length) {
      html += '<p class="text-sm text-slate-500">此隊伍尚未提交答案。</p>';
    } else {
      subs.forEach(function (s) {
        const loc = locs.find(function (l) {
          return l.id === s.locationId;
        });
        const qMap = {};
        ((loc && loc.questions) || []).forEach(function (q) {
          qMap[q.id] = q;
        });
        let detail = "";
        const ans = s.answers || {};
        const keys = Object.keys(ans);
        if (loc && loc.questions && loc.questions.length) {
          loc.questions
            .slice()
            .sort(function (a, b) {
              return (a.order || 0) - (b.order || 0);
            })
            .forEach(function (q) {
              const val = ans[q.id];
              if (val == null || String(val).trim() === "") return;
              detail +=
                '<div class="pl-2 border-l-2 border-emerald-200 my-1">' +
                '<div class="text-slate-600">' +
                esc(q.text) +
                '</div><div class="font-medium text-slate-800">' +
                esc(val) +
                "</div></div>";
            });
        } else {
          keys.forEach(function (k) {
            if (!(ans[k] || "").toString().trim()) return;
            const label = qMap[k] ? qMap[k].text : k;
            detail +=
              '<div class="pl-2 border-l-2 border-emerald-200 my-1">' +
              '<div class="text-slate-600">' +
              esc(label) +
              '</div><div class="font-medium text-slate-800">' +
              esc(ans[k]) +
              "</div></div>";
          });
        }
        html +=
          '<div class="bg-white border rounded-xl p-3 text-xs space-y-1 mb-2">' +
          '<div class="font-semibold text-sm text-slate-800">' +
          esc(loc ? loc.name : s.locationId) +
          "</div>" +
          '<div class="text-slate-500">' +
          (s.submittedAt
            ? new Date(s.submittedAt).toLocaleString("zh-HK")
            : "") +
          (s.coords
            ? " · 位置 " +
              Number(s.coords.lat).toFixed(5) +
              ", " +
              Number(s.coords.lng).toFixed(5)
            : "") +
          "</div>" +
          (detail || '<div class="text-slate-400">（無答案內容）</div>') +
          "</div>";
      });
    }
    panel.innerHTML = html;
    panel.classList.remove("hidden");
    const closeBtn = document.getElementById("btnCloseMonDetail");
    if (closeBtn) {
      closeBtn.onclick = function () {
        panel.classList.add("hidden");
        panel.innerHTML = "";
      };
    }
    panel.scrollIntoView({ behavior: "smooth", block: "nearest" });
  }

  async function openMonitor(projectId) {
    StudentUI.showLoading(true);
    try {
      const authed = await Auth.ensureTeacherAuth();
      if (!authed) {
        alert("登入狀態已過期或未還原，請重新以 Google 登入後再監察。");
        Auth.clearSession();
        if (typeof window.showLandingFromTeacher === "function") {
          window.showLandingFromTeacher();
        } else {
          document.getElementById("appHeader").classList.add("hidden");
          StudentUI.showView("view-landing");
          window.location.hash = "#/";
        }
        return;
      }
      session = authed;
      monitorProjectId = projectId;
      currentProject = await DataStore.getProject(projectId);
      locations = await DataStore.listLocations(projectId);
      teams = await DataStore.listTeams(projectId);
      const sessions = await DataStore.listTeamSessions(projectId);
      const submissions = await DataStore.listSubmissions(projectId);
      const sortedLocs = locations
        .slice()
        .sort(function (a, b) {
          return (a.order || 0) - (b.order || 0);
        });
      monitorCache = {
        project: currentProject,
        locations: sortedLocs,
        teams: teams,
        sessions: sessions,
        submissions: submissions
      };
      showTeacherSection("monitor");
      document.getElementById("monTitle").textContent =
        "監察：" + (currentProject.title || "");
      const detail = document.getElementById("monTeamDetail");
      if (detail) {
        detail.classList.add("hidden");
        detail.innerHTML = "";
      }
      const sessBox = document.getElementById("monSessions");
      sessBox.innerHTML = "";
      const sessMap = {};
      sessions.forEach(function (s) {
        sessMap[s.teamId] = s;
      });
      if (!teams.length && !sessions.length) {
        sessBox.innerHTML =
          '<p class="text-sm text-slate-500">尚無隊伍。請先在編輯頁新增隊伍。</p>';
      }
      teams.forEach(function (t) {
        const s = sessMap[t.id] || {};
        const card = document.createElement("button");
        card.type = "button";
        card.className =
          "w-full text-left bg-white border border-slate-200 rounded-xl p-3 text-sm hover:border-emerald-400 hover:shadow-sm transition";
        card.innerHTML = buildTeamProgressHtml(t, s, sortedLocs);
        card.onclick = function () {
          renderTeamDetail(t.id);
        };
        sessBox.appendChild(card);
      });
      sessions.forEach(function (s) {
        if (teams.some(function (t) {
          return t.id === s.teamId;
        }))
          return;
        const card = document.createElement("button");
        card.type = "button";
        card.className =
          "w-full text-left bg-white border border-amber-200 rounded-xl p-3 text-sm";
        card.innerHTML = buildTeamProgressHtml(
          { id: s.teamId, name: s.teamId },
          s,
          sortedLocs
        );
        card.onclick = function () {
          renderTeamDetail(s.teamId);
        };
        sessBox.appendChild(card);
      });

      window._monExport = {
        project: currentProject,
        locations: sortedLocs,
        submissions: submissions
      };
      window.location.hash = "#/teacher/monitor/" + projectId;
    } catch (e) {
      if (Auth.isPermissionError && Auth.isPermissionError(e)) {
        alert(
          "無法讀取監察資料（權限不足）。通常是 Google 登入尚未還原。請登出後重新以 Google 登入，再按監察。"
        );
      } else {
        alert(e.message || String(e));
      }
    } finally {
      StudentUI.showLoading(false);
    }
  }

  async function refreshMonitor() {
    if (!monitorProjectId) {
      alert("請先從專案列表開啟監察。");
      return;
    }
    await openMonitor(monitorProjectId);
  }

  function exportCSV() {
    const data = window._monExport;
    if (!data || !data.submissions.length) {
      alert("沒有可匯出的資料");
      return;
    }
    const csv = DataStore.exportSubmissionsCSV(
      data.project,
      data.locations,
      data.submissions
    );
    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
    const a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download =
      (data.project.slug || "export") +
      "_" +
      new Date().toISOString().slice(0, 10) +
      ".csv";
    a.click();
  }

  function backToProjects() {
    showTeacherSection("projects");
    refreshProjects();
    window.location.hash = "#/teacher";
  }

  return {
    start,
    createNewProject,
    openEditor,
    openMonitor,
    refreshMonitor,
    saveSettings,
    switchEditorTab,
    openLocModal,
    closeLocModal,
    saveLocModal,
    addQuestionRow,
    addTeam,
    exportCSV,
    backToProjects,
    refreshProjects
  };
})();
