  function destroyTrailMap() {
    if (trailMapInstance) {
      try {
        trailMapInstance.remove();
      } catch (e) {}
      trailMapInstance = null;
    }
  }

  function renderTrailMap(trail, locs, sess) {
    destroyTrailMap();
    const el = document.getElementById("monTrailMap");
    if (!el || typeof L === "undefined") {
      if (el) {
        el.innerHTML =
          '<p class="text-xs text-slate-500 p-3 text-center">地圖元件未載入</p>';
      }
      return;
    }
    const pts = (trail || [])
      .filter(function (p) {
        return p && p.lat != null && p.lng != null;
      })
      .map(function (p) {
        return [Number(p.lat), Number(p.lng)];
      });
    if (!pts.length && sess && sess.lastLat != null) {
      pts.push([Number(sess.lastLat), Number(sess.lastLng)]);
    }
    const map = L.map(el, { zoomControl: true, attributionControl: true });
    trailMapInstance = map;
    // LandsD GeoInfo Map (public XYZ) — basemap + TC labels
    L.tileLayer(
      "https://mapapi.geodata.gov.hk/gs/api/v1.0.0/xyz/basemap/WGS84/{z}/{x}/{y}.png",
      {
        minZoom: 10,
        maxZoom: 20,
        attribution: "地圖資料 © 地政總署 Lands Department"
      }
    ).addTo(map);
    L.tileLayer(
      "https://mapapi.geodata.gov.hk/gs/api/v1.0.0/xyz/label/hk/tc/WGS84/{z}/{x}/{y}.png",
      {
        minZoom: 10,
        maxZoom: 20,
        attribution: ""
      }
    ).addTo(map);
    const fitPts = pts.slice();
    if (pts.length) {
      L.polyline(pts, {
        color: "#059669",
        weight: 4,
        opacity: 0.85
      }).addTo(map);
      L.circleMarker(pts[pts.length - 1], {
        radius: 7,
        color: "#047857",
        fillColor: "#34d399",
        fillOpacity: 1
      })
        .bindPopup("目前／最新位置")
        .addTo(map);
      if (pts.length > 1) {
        L.circleMarker(pts[0], {
          radius: 5,
          color: "#64748b",
          fillColor: "#cbd5e1",
          fillOpacity: 1
        })
          .bindPopup("起點")
          .addTo(map);
      }
    }
    const checkpointIcon = L.icon({
      iconUrl: "https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon.png",
      iconRetinaUrl:
        "https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon-2x.png",
      shadowUrl: "https://unpkg.com/leaflet@1.9.4/dist/images/marker-shadow.png",
      iconSize: [12.5, 20.5],
      iconAnchor: [6.25, 20.5],
      popupAnchor: [1, -17],
      shadowSize: [20.5, 20.5],
      shadowAnchor: [6.25, 20.5]
    });
    (locs || []).forEach(function (loc, i) {
      if (loc.lat == null || loc.lng == null) return;
      const ll = [Number(loc.lat), Number(loc.lng)];
      fitPts.push(ll);
      L.marker(ll, { icon: checkpointIcon })
        .bindPopup(
          "<strong>" + esc(loc.name) + "</strong><br>檢查點 " + (i + 1)
        )
        .addTo(map);
    });
    if (fitPts.length) {
      map.fitBounds(fitPts, { padding: [28, 28], maxZoom: 17 });
    } else {
      map.setView([22.21, 114.03], 13);
    }
    setTimeout(function () {
      map.invalidateSize();
    }, 100);
  }

  async function openMonitor(projectId) {
    StudentUI.showLoading(true);
    try {
      const authed = await Auth.ensureTeacherAuth();
      if (!authed) {
        alert("登入狀態已過期或未還原，請重新以 Google 登入後再監察。");
        Auth.clearSession();
        if (typeof window.showLandingFromTeacher === "function") {
          window.showLandingFromTeacher();
        } else {
          document.getElementById("appHeader").classList.add("hidden");
          StudentUI.showView("view-landing");
          window.location.hash = "#/";
        }
        return;
      }
      session = authed;
      monitorProjectId = projectId;
      currentProject = await DataStore.getProject(projectId);
      locations = await DataStore.listLocations(projectId);
      teams = await DataStore.listTeams(projectId);
      const sessions = await DataStore.listTeamSessions(projectId);
      const submissions = await DataStore.listSubmissions(projectId);
      const sortedLocs = locations
        .slice()
        .sort(function (a, b) {
          return (a.order || 0) - (b.order || 0);
        });
      monitorCache = {
        project: currentProject,
        locations: sortedLocs,
        teams: teams,
        sessions: sessions,
        submissions: submissions
      };
      showTeacherSection("monitor");
      document.getElementById("monTitle").textContent =
        "監察：" + (currentProject.title || "");
      destroyTrailMap();
      const detail = document.getElementById("monTeamDetail");
      if (detail) {
        detail.classList.add("hidden");
        detail.innerHTML = "";
      }
      const sessBox = document.getElementById("monSessions");
      sessBox.innerHTML = "";
      const sessMap = {};
      sessions.forEach(function (s) {
        sessMap[s.teamId] = s;
      });
      if (!teams.length && !sessions.length) {
        sessBox.innerHTML =
          '<p class="text-sm text-slate-500">尚無隊伍。請先在編輯頁新增隊伍。</p>';
      }
      teams.forEach(function (t) {
        const s = sessMap[t.id] || {};
        const card = document.createElement("button");
        card.type = "button";
        card.className =
          "w-full text-left bg-white border border-slate-200 rounded-xl p-3 text-sm hover:border-emerald-400 hover:shadow-sm transition";
        card.innerHTML = buildTeamProgressHtml(t, s, sortedLocs);
        card.onclick = function () {
          renderTeamDetail(t.id);
        };
        sessBox.appendChild(card);
      });
      sessions.forEach(function (s) {
        if (teams.some(function (t) {
          return t.id === s.teamId;
        }))
          return;
        const card = document.createElement("button");
        card.type = "button";
        card.className =
          "w-full text-left bg-white border border-amber-200 rounded-xl p-3 text-sm";
        card.innerHTML = buildTeamProgressHtml(
          { id: s.teamId, name: s.teamId },
          s,
          sortedLocs
        );
        card.onclick = function () {
          renderTeamDetail(s.teamId);
        };
        sessBox.appendChild(card);
      });

      window._monExport = {
        project: currentProject,
        locations: sortedLocs,
        submissions: submissions
      };
      window.location.hash = "#/teacher/monitor/" + projectId;
    } catch (e) {
      if (Auth.isPermissionError && Auth.isPermissionError(e)) {
        alert(
          "無法讀取監察資料（權限不足）。通常是 Google 登入尚未還原。請登出後重新以 Google 登入，再按監察。"
        );
      } else {
        alert(e.message || String(e));
      }
    } finally {
      StudentUI.showLoading(false);
    }
  }

  async function refreshMonitor() {
    if (!monitorProjectId) {
      alert("請先從專案列表開啟監察。");
      return;
    }
    await openMonitor(monitorProjectId);
  }

  /**
   * Clear runtime data: submissions + teamSessions progress/GPS trail.
   * Keeps locations, questions, teams, project settings.
   */
  async function clearProjectData() {
    const pid = monitorProjectId || (currentProject && currentProject.id);
    if (!pid) {
      alert("請先開啟監察或編輯頁的專案。");
      return;
    }
    const okAuth = await requireTeacherAuth("清除資料");
    if (!okAuth) return;
    const title =
      (currentProject && currentProject.title) ||
      (monitorCache && monitorCache.project && monitorCache.project.title) ||
      pid;
    const yes = confirm(
      "確定清除「" +
        title +
        "」的運行資料？\n\n將刪除：所有提交答案、隊伍進度、GPS 軌跡與解鎖狀態。\n保留：地區／問題／隊伍帳號／專案設定。\n\n此操作不可復原。"
    );
    if (!yes) return;
    const yes2 = confirm("再確認一次：真的要清除此專案的提交與 GPS 資料？");
    if (!yes2) return;
    StudentUI.showLoading(true);
    try {
      const result = await withTimeout(
        DataStore.clearProjectRuntimeData(pid),
        30000,
        "清除資料"
      );
      alert(
        "已清除。提交：" +
          (result.submissionsCleared != null
            ? result.submissionsCleared
            : "完成") +
          "；隊伍進度重設：" +
          (result.sessionsReset != null ? result.sessionsReset : "完成")
      );
      if (monitorProjectId === pid) await openMonitor(pid);
    } catch (e) {
      alert(writeErrorMessage(e, "清除失敗"));
    } finally {
      StudentUI.showLoading(false);
    }
  }


  function downloadJsonFile(filename, obj) {
    const blob = new Blob([JSON.stringify(obj, null, 2)], {
      type: "application/json;charset=utf-8"
    });
    const a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download = filename;
    a.click();
    setTimeout(function () {
      try {
        URL.revokeObjectURL(a.href);
      } catch (e) {}
    }, 1500);
  }

  async function exportLocationsJson() {
    if (!currentProject) {
      alert("請先開啟專案");
      return;
    }
    const authed = await requireTeacherAuth("匯出地區");
    if (!authed) return;
    const locs = sortedLocations().map(function (loc) {
      return {
        id: loc.id,
        name: loc.name,
        lat: loc.lat,
        lng: loc.lng,
        description: loc.description || "",
        order: loc.order || 1,
        images: loc.images || [],
        questions: loc.questions || []
      };
    });
    const payload = {
      format: "gps-field-game-locations",
      version: 1,
      projectSlug: currentProject.slug || "",
      projectTitle: currentProject.title || "",
      exportedAt: new Date().toISOString(),
      locations: locs
    };
    const slug = (currentProject.slug || "project").replace(/[^\w\-]+/g, "_");
    downloadJsonFile("locations-" + slug + ".json", payload);
  }

  function parseLocationsImport(raw) {
    let data = raw;
    if (typeof raw === "string") {
      data = JSON.parse(raw);
    }
    let list = null;
    if (Array.isArray(data)) list = data;
    else if (data && Array.isArray(data.locations)) list = data.locations;
    else throw new Error("JSON 格式不正確：需要 locations 陣列");
    return list.map(function (loc, i) {
      if (!loc || typeof loc !== "object") {
        throw new Error("第 " + (i + 1) + " 筆地區無效");
      }
      return {
        id: loc.id || null,
        name: loc.name || "新地區",
        lat: Number(loc.lat) || 0,
        lng: Number(loc.lng) || 0,
        description: loc.description || "",
        order: loc.order != null ? Number(loc.order) : i + 1,
        images: Array.isArray(loc.images) ? loc.images : [],
        questions: Array.isArray(loc.questions) ? loc.questions : []
      };
    });
  }

  async function importLocationsJson(fileOrText) {
    if (!currentProject) {
      alert("請先開啟專案");
      return;
    }
    const authed = await requireTeacherAuth("匯入地區");
    if (!authed) return;
    let text = "";
    try {
      if (typeof fileOrText === "string") {
        text = fileOrText;
      } else if (fileOrText && typeof fileOrText.text === "function") {
        text = await fileOrText.text();
      } else {
        text = prompt("請貼上地區 JSON（或先選檔案）：", "") || "";
      }
      if (!text.trim()) return;
      const incoming = parseLocationsImport(text);
      if (!incoming.length) {
        alert("檔案沒有地區資料");
        return;
      }
      if (
        !confirm(
          "即將匯入 " +
            incoming.length +
            " 個地區／問題（更新或新增）。確定繼續？"
        )
      ) {
        return;
      }
      const doReplace = confirm(
        "是否完整取代？\n確定＝刪除檔案中沒有的現有地區；取消＝僅更新／新增並保留其餘地區。"
      );
      StudentUI.showLoading(true);
      try {
        const kept = new Set();
        for (let i = 0; i < incoming.length; i++) {
          const loc = incoming[i];
          const saved = await withTimeout(
            DataStore.upsertLocation(currentProject.id, loc.id || null, loc),
            20000,
            "匯入地區"
          );
          kept.add(saved.id);
        }
        if (doReplace) {
          for (let j = 0; j < locations.length; j++) {
            const old = locations[j];
            if (!kept.has(old.id)) {
              await withTimeout(
                DataStore.deleteLocation(currentProject.id, old.id),
                15000,
                "刪除舊地區"
              );
            }
          }
        }
        locations = await withTimeout(
          DataStore.listLocations(currentProject.id),
          15000,
          "重新載入地區"
        );
        renderLocList();
        alert("匯入完成（" + locations.length + " 個地區）");
      } catch (e) {
        alert(writeErrorMessage(e));
      } finally {
        StudentUI.showLoading(false);
      }
    } catch (e) {
      alert("無法解析 JSON：" + (e && e.message ? e.message : e));
    }
  }

  function exportCSV() {
    const data = window._monExport;
    if (!data || !data.submissions.length) {
      alert("沒有可匯出的資料");
      return;
    }
    const csv = DataStore.exportSubmissionsCSV(
      data.project,
      data.locations,
      data.submissions
    );
    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
    const a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download =
      (data.project.slug || "export") +
      "_" +
      new Date().toISOString().slice(0, 10) +
      ".csv";
    a.click();
  }
