  function destroyTrailMap() {
    if (trailMapInstance) {
      try {
        trailMapInstance.remove();
      } catch (e) {}
      trailMapInstance = null;
    }
  }

  function renderTrailMap(trail, locs, sess) {
    destroyTrailMap();
    const el = document.getElementById("monTrailMap");
    if (!el || typeof L === "undefined") {
      if (el) {
        el.innerHTML =
          '<p class="text-xs text-slate-500 p-3 text-center">地圖元件未載入</p>';
      }
      return;
    }
    const pts = (trail || [])
      .filter(function (p) {
        return p && p.lat != null && p.lng != null;
      })
      .map(function (p) {
        return [Number(p.lat), Number(p.lng)];
      });
    if (!pts.length && sess && sess.lastLat != null) {
      pts.push([Number(sess.lastLat), Number(sess.lastLng)]);
    }
    const map = L.map(el, { zoomControl: true, attributionControl: true });
    trailMapInstance = map;
    L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png", {
      maxZoom: 19,
      attribution: "(c) OpenStreetMap"
    }).addTo(map);
    const fitPts = pts.slice();
    if (pts.length) {
      L.polyline(pts, {
        color: "#059669",
        weight: 4,
        opacity: 0.85
      }).addTo(map);
      L.circleMarker(pts[pts.length - 1], {
        radius: 7,
        color: "#047857",
        fillColor: "#34d399",
        fillOpacity: 1
      })
        .bindPopup("目前／最新位置")
        .addTo(map);
      if (pts.length > 1) {
        L.circleMarker(pts[0], {
          radius: 5,
          color: "#64748b",
          fillColor: "#cbd5e1",
          fillOpacity: 1
        })
          .bindPopup("起點")
          .addTo(map);
      }
    }
    (locs || []).forEach(function (loc, i) {
      if (loc.lat == null || loc.lng == null) return;
      const ll = [Number(loc.lat), Number(loc.lng)];
      fitPts.push(ll);
      L.marker(ll)
        .bindPopup(
          "<strong>" + esc(loc.name) + "</strong><br>檢查點 " + (i + 1)
        )
        .addTo(map);
    });
    if (fitPts.length) {
      map.fitBounds(fitPts, { padding: [28, 28], maxZoom: 17 });
    } else {
      map.setView([22.21, 114.03], 13);
    }
    setTimeout(function () {
      map.invalidateSize();
    }, 100);
  }

  async function openMonitor(projectId) {
    StudentUI.showLoading(true);
    try {
      const authed = await Auth.ensureTeacherAuth();
      if (!authed) {
        alert("登入狀態已過期或未還原，請重新以 Google 登入後再監察。");
        Auth.clearSession();
        if (typeof window.showLandingFromTeacher === "function") {
          window.showLandingFromTeacher();
        } else {
          document.getElementById("appHeader").classList.add("hidden");
          StudentUI.showView("view-landing");
          window.location.hash = "#/";
        }
        return;
      }
      session = authed;
      monitorProjectId = projectId;
      currentProject = await DataStore.getProject(projectId);
      locations = await DataStore.listLocations(projectId);
      teams = await DataStore.listTeams(projectId);
      const sessions = await DataStore.listTeamSessions(projectId);
      const submissions = await DataStore.listSubmissions(projectId);
      const sortedLocs = locations
        .slice()
        .sort(function (a, b) {
          return (a.order || 0) - (b.order || 0);
        });
      monitorCache = {
        project: currentProject,
        locations: sortedLocs,
        teams: teams,
        sessions: sessions,
        submissions: submissions
      };
      showTeacherSection("monitor");
      document.getElementById("monTitle").textContent =
        "監察：" + (currentProject.title || "");
      destroyTrailMap();
      const detail = document.getElementById("monTeamDetail");
      if (detail) {
        detail.classList.add("hidden");
        detail.innerHTML = "";
      }
      const sessBox = document.getElementById("monSessions");
      sessBox.innerHTML = "";
      const sessMap = {};
      sessions.forEach(function (s) {
        sessMap[s.teamId] = s;
      });
      if (!teams.length && !sessions.length) {
        sessBox.innerHTML =
          '<p class="text-sm text-slate-500">尚無隊伍。請先在編輯頁新增隊伍。</p>';
      }
      teams.forEach(function (t) {
        const s = sessMap[t.id] || {};
        const card = document.createElement("button");
        card.type = "button";
        card.className =
          "w-full text-left bg-white border border-slate-200 rounded-xl p-3 text-sm hover:border-emerald-400 hover:shadow-sm transition";
        card.innerHTML = buildTeamProgressHtml(t, s, sortedLocs);
        card.onclick = function () {
          renderTeamDetail(t.id);
        };
        sessBox.appendChild(card);
      });
      sessions.forEach(function (s) {
        if (teams.some(function (t) {
          return t.id === s.teamId;
        }))
          return;
        const card = document.createElement("button");
        card.type = "button";
        card.className =
          "w-full text-left bg-white border border-amber-200 rounded-xl p-3 text-sm";
        card.innerHTML = buildTeamProgressHtml(
          { id: s.teamId, name: s.teamId },
          s,
          sortedLocs
        );
        card.onclick = function () {
          renderTeamDetail(s.teamId);
        };
        sessBox.appendChild(card);
      });

      window._monExport = {
        project: currentProject,
        locations: sortedLocs,
        submissions: submissions
      };
      window.location.hash = "#/teacher/monitor/" + projectId;
    } catch (e) {
      if (Auth.isPermissionError && Auth.isPermissionError(e)) {
        alert(
          "無法讀取監察資料（權限不足）。通常是 Google 登入尚未還原。請登出後重新以 Google 登入，再按監察。"
        );
      } else {
        alert(e.message || String(e));
      }
    } finally {
      StudentUI.showLoading(false);
    }
  }

  async function refreshMonitor() {
    if (!monitorProjectId) {
      alert("請先從專案列表開啟監察。");
      return;
    }
    await openMonitor(monitorProjectId);
  }

  /**
   * Clear runtime data: submissions + teamSessions progress/GPS trail.
   * Keeps locations, questions, teams, project settings.
   */
  async function clearProjectData() {
    const pid = monitorProjectId || (currentProject && currentProject.id);
    if (!pid) {
      alert("請先開啟監察或編輯頁的專案。");
      return;
    }
    const okAuth = await requireTeacherAuth("清除資料");
    if (!okAuth) return;
    const title =
      (currentProject && currentProject.title) ||
      (monitorCache && monitorCache.project && monitorCache.project.title) ||
      pid;
    const yes = confirm(
      "確定清除「" +
        title +
        "」的運行資料？\n\n將刪除：所有提交答案、隊伍進度、GPS 軌跡與解鎖狀態。\n保留：地區／問題／隊伍帳號／專案設定。\n\n此操作不可復原。"
    );
    if (!yes) return;
    const yes2 = confirm("再確認一次：真的要清除此專案的提交與 GPS 資料？");
    if (!yes2) return;
    StudentUI.showLoading(true);
    try {
      const result = await withTimeout(
        DataStore.clearProjectRuntimeData(pid),
        30000,
        "清除資料"
      );
      alert(
        "已清除。提交：" +
          (result.submissionsCleared != null
            ? result.submissionsCleared
            : "完成") +
          "；隊伍進度重設：" +
          (result.sessionsReset != null ? result.sessionsReset : "完成")
      );
      if (monitorProjectId === pid) await openMonitor(pid);
    } catch (e) {
      alert(writeErrorMessage(e, "清除失敗"));
    } finally {
      StudentUI.showLoading(false);
    }
  }

  function exportCSV() {
    const data = window._monExport;
    if (!data || !data.submissions.length) {
      alert("沒有可匯出的資料");
      return;
    }
    const csv = DataStore.exportSubmissionsCSV(
      data.project,
      data.locations,
      data.submissions
    );
    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
    const a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download =
      (data.project.slug || "export") +
      "_" +
      new Date().toISOString().slice(0, 10) +
      ".csv";
    a.click();
  }

  function backToProjects() {
    showTeacherSection("projects");
    refreshProjects();
    window.location.hash = "#/teacher";
  }

    addQuestionRow,
    addTeam,

  };
