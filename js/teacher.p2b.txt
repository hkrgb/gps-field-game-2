
  function buildTeamProgressHtml(t, s, locs) {
    const doneIds = new Set(s.completedLocationIds || []);
    const done = doneIds.size;
    const total = locs.length;
    const pct = total ? Math.round((done / total) * 100) : 0;
    let checks = "";
    locs.forEach(function (loc) {
      const ok = doneIds.has(loc.id);
      checks +=
        '<span class="inline-flex items-center gap-0.5 text-[10px] px-1.5 py-0.5 rounded ' +
        (ok ? "bg-emerald-100 text-emerald-800" : "bg-slate-100 text-slate-500") +
        '" title="' +
        esc(loc.name) +
        '">' +
        (ok ? "✓" : "○") +
        " " +
        esc(loc.name) +
        "</span>";
    });
    let gpsLine = '<span class="text-slate-400">尚未回報位置</span>';
    if (s.lastLat != null && s.lastLng != null) {
      const info = nearestCheckpointInfo(s.lastLat, s.lastLng, locs);
      if (info) {
        gpsLine =
          '<span class="text-emerald-800"><i class="fa-solid fa-location-crosshairs mr-1"></i>' +
          esc(info.phrase) +
          "</span>";
      } else {
        gpsLine =
          '<span class="text-slate-600">位置 ' +
          Number(s.lastLat).toFixed(5) +
          ", " +
          Number(s.lastLng).toFixed(5) +
          "</span>";
      }
      if (s.lastGpsAt) {
        gpsLine +=
          '<span class="text-slate-400 ml-1">· ' +
          new Date(s.lastGpsAt).toLocaleString("zh-HK") +
          "</span>";
      }
    }
    return (
      '<div class="font-bold text-slate-800 flex justify-between items-center gap-2">' +
      "<span>" +
      esc(t.name) +
      '</span><span class="text-xs font-semibold text-emerald-700">完成 ' +
      done +
      "/" +
      total +
      "</span></div>" +
      '<div class="mt-1.5 h-1.5 bg-slate-100 rounded-full overflow-hidden">' +
      '<div class="h-full bg-emerald-500 rounded-full" style="width:' +
      pct +
      '%"></div></div>' +
      '<div class="mt-2 flex flex-wrap gap-1">' +
      (checks || '<span class="text-[10px] text-slate-400">未設定地區</span>') +
      "</div>" +
      '<div class="mt-2 text-[11px] leading-snug">' +
      gpsLine +
      "</div>" +
      '<div class="text-[10px] text-slate-400 mt-1">上次登入：' +
      (s.lastLoginAt
        ? new Date(s.lastLoginAt).toLocaleString("zh-HK")
        : "尚未登入") +
      " · 點擊查看答題</div>"
    );
  }

  function renderTeamDetail(teamId) {
    const panel = document.getElementById("monTeamDetail");
    if (!panel || !monitorCache) return;
    const t =
      monitorCache.teams.find(function (x) {
        return x.id === teamId;
      }) || { id: teamId, name: teamId };
    const locs = monitorCache.locations;
    const sess =
      (monitorCache.sessions || []).find(function (s) {
        return s.teamId === teamId;
      }) || {};
    const subs = (monitorCache.submissions || []).filter(function (s) {
      return s.teamId === teamId;
    });
    const trail = Array.isArray(sess.trail) ? sess.trail.slice() : [];
    const unlockedMap = sess.unlockedEdits || {};

    let html =
      '<div class="flex justify-between items-center gap-2 mb-2">' +
      '<h4 class="font-bold text-sm text-slate-800">隊伍詳情：' +
      esc(t.name) +
      "</h4>" +
      '<button type="button" id="btnCloseMonDetail" class="text-xs text-slate-500 hover:underline">關閉</button></div>';

    /* Timeline */
    html +=
      '<div class="mb-3">' +
      '<div class="text-xs font-bold text-slate-700 mb-1.5"><i class="fa-solid fa-timeline mr-1 text-emerald-600"></i>GPS 時間軸</div>';
    if (!trail.length && sess.lastLat == null) {
      html += '<p class="text-xs text-slate-500">尚無 GPS 軌跡。</p>';
    } else {
      const points = trail.length
        ? trail.slice().reverse()
        : [
            {
              lat: sess.lastLat,
              lng: sess.lastLng,
              at: sess.lastGpsAt
            }
          ];
      html += '<ul class="space-y-1.5 max-h-40 overflow-y-auto text-[11px]">';
      points.slice(0, 40).forEach(function (pt) {
        if (pt.lat == null || pt.lng == null) return;
        const info = nearestCheckpointInfo(pt.lat, pt.lng, locs);
        const when = pt.at
          ? new Date(pt.at).toLocaleString("zh-HK")
          : "—";
        html +=
          '<li class="flex gap-2 border-l-2 border-emerald-300 pl-2">' +
          '<span class="text-slate-400 shrink-0 w-[7.5rem]">' +
          esc(when) +
          "</span>" +
          '<span class="text-slate-700">' +
          (info
            ? esc(info.phrase)
            : Number(pt.lat).toFixed(5) + ", " + Number(pt.lng).toFixed(5)) +
          "</span></li>";
      });
      html += "</ul>";
      if (points.length > 40) {
        html +=
          '<p class="text-[10px] text-slate-400 mt-1">顯示最近 40 點（共 ' +
          points.length +
          "）</p>";
      }
    }
    html += "</div>";

    /* Map */
    html +=
      '<div class="mb-3">' +
      '<div class="text-xs font-bold text-slate-700 mb-1.5"><i class="fa-solid fa-route mr-1 text-blue-600"></i>步行路徑</div>' +
      '<div id="monTrailMap" class="mon-trail-map rounded-xl border border-slate-200 overflow-hidden bg-slate-100"></div>' +
      '<p class="text-[10px] text-slate-400 mt-1">綠線＝GPS 軌跡；圖釘＝檢查點</p></div>';

    /* Unlock controls */
    html +=
      '<div class="mb-3">' +
      '<div class="text-xs font-bold text-slate-700 mb-1.5"><i class="fa-solid fa-lock-open mr-1 text-amber-600"></i>解鎖再編輯</div>' +
      '<p class="text-[10px] text-slate-500 mb-1.5">提交後預設鎖定。可解鎖指定地區讓該隊修改一次（再提交後會再鎖）。</p>' +
      '<div class="flex flex-wrap gap-1.5" id="monUnlockBtns">';
    locs.forEach(function (loc) {
      const done = (sess.completedLocationIds || []).indexOf(loc.id) >= 0;
      const isUn = !!unlockedMap[loc.id];
      if (!done) return;
      html +=
        '<button type="button" data-unlock-loc="' +
        esc(loc.id) +
        '" data-unlock-on="' +
        (isUn ? "0" : "1") +
        '" class="text-[11px] px-2 py-1 rounded-lg border ' +
        (isUn
          ? "bg-amber-100 border-amber-300 text-amber-900"
          : "bg-white border-slate-200 text-slate-700") +
        '">' +
        (isUn ? "取消解鎖 · " : "解鎖 · ") +
        esc(loc.name) +
        "</button>";
    });
    html += "</div></div>";

    /* Answers */
    html +=
      '<div class="text-xs font-bold text-slate-700 mb-1.5"><i class="fa-solid fa-clipboard-list mr-1 text-emerald-600"></i>答題內容</div>';
    if (!subs.length) {
      html += '<p class="text-sm text-slate-500">此隊伍尚未提交答案。</p>';
    } else {
      subs.forEach(function (s) {
        const loc = locs.find(function (l) {
          return l.id === s.locationId;
        });
        const qMap = {};
        ((loc && loc.questions) || []).forEach(function (q) {
          qMap[q.id] = q;
        });
        let detail = "";
        const ans = s.answers || {};
        const keys = Object.keys(ans);
        if (loc && loc.questions && loc.questions.length) {
          loc.questions
            .slice()
            .sort(function (a, b) {
              return (a.order || 0) - (b.order || 0);
            })
            .forEach(function (q) {
              const val = ans[q.id];
              if (val == null || String(val).trim() === "") return;
              detail +=
                '<div class="pl-2 border-l-2 border-emerald-200 my-1">' +
                '<div class="text-slate-600">' +
                esc(q.text) +
                '</div><div class="font-medium text-slate-800">' +
                esc(val) +
                "</div></div>";
            });
        } else {
          keys.forEach(function (k) {
            if (!(ans[k] || "").toString().trim()) return;
            const label = qMap[k] ? qMap[k].text : k;
            detail +=
              '<div class="pl-2 border-l-2 border-emerald-200 my-1">' +
              '<div class="text-slate-600">' +
              esc(label) +
              '</div><div class="font-medium text-slate-800">' +
              esc(ans[k]) +
              "</div></div>";
          });
        }
        html +=
          '<div class="bg-white border rounded-xl p-3 text-xs space-y-1 mb-2">' +
          '<div class="font-semibold text-sm text-slate-800">' +
          esc(loc ? loc.name : s.locationId) +
          "</div>" +
          '<div class="text-slate-500">' +
          (s.submittedAt
            ? new Date(s.submittedAt).toLocaleString("zh-HK")
            : "") +
          (s.coords
            ? " · 位置 " +
              Number(s.coords.lat).toFixed(5) +
              ", " +
              Number(s.coords.lng).toFixed(5)
            : "") +
          "</div>" +
          (detail || '<div class="text-slate-400">（無答案內容）</div>') +
          "</div>";
      });
    }
    panel.innerHTML = html;
    panel.classList.remove("hidden");
    const closeBtn = document.getElementById("btnCloseMonDetail");
    if (closeBtn) {
      closeBtn.onclick = function () {
        destroyTrailMap();
        panel.classList.add("hidden");
        panel.innerHTML = "";
      };
    }
    panel.querySelectorAll("[data-unlock-loc]").forEach(function (btn) {
      btn.onclick = async function () {
        const locId = btn.getAttribute("data-unlock-loc");
        const on = btn.getAttribute("data-unlock-on") === "1";
        const ok = await requireTeacherAuth(on ? "解鎖" : "取消解鎖");
        if (!ok) return;
        StudentUI.showLoading(true);
        try {
          await DataStore.setLocationEditUnlock(
            monitorProjectId,
            teamId,
            locId,
            on
          );
          await openMonitor(monitorProjectId);
          renderTeamDetail(teamId);
        } catch (e) {
          alert(writeErrorMessage(e));
        } finally {
          StudentUI.showLoading(false);
        }
      };
    });
    setTimeout(function () {
      renderTrailMap(trail, locs, sess);
    }, 40);
    panel.scrollIntoView({ behavior: "smooth", block: "nearest" });
  }
