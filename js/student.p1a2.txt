
  function renderProgressSquares() {
    const el = document.getElementById("progressSquares");
    if (!el) return;
    el.innerHTML = "";
    locations.forEach((loc, i) => {
      const done = isLocComplete(loc.id);
      const sq = document.createElement("div");
      sq.className =
        "progress-square " + (done ? "filled" : "empty");
      sq.title = loc.name + (done ? "（已完成）" : "（未完成）");
      sq.textContent = String(i + 1);
      el.appendChild(sq);
    });
  }

  function syncTestModeFromProject() {
    testMode = !!(project && project.showTestMode);
  }

  function updateHeader() {
    const nameEl = document.getElementById("headerUserName");
    if (nameEl) nameEl.textContent = session ? session.teamName : "";
    renderProgressSquares();
    const btn = document.getElementById("btnCompletion");
    if (btn) {
      const allDone = !!(session && isAllComplete());
      btn.classList.toggle("hidden", !allDone);
      btn.disabled = !allDone;
    }
  }

  async function start(sess, proj) {
    session = sess;
    project = proj;
    locations = await DataStore.listLocations(project.id);
    teamSession = await DataStore.getTeamSession(project.id, session.teamId);
    lastSessionFetchAt = Date.now();
    loadDrafts();
    reconcileDraftsWithServer(null, teamSession);
    applyFontSize();
    syncTestModeFromProject();
    updateHeader();
    bindSessionRefreshListeners();
    startSessionRefreshInterval();
    await showHub({ skipRefresh: true });
    refreshGPS();
    startGpsWatch();
    startGpsAutoInterval();
    if (window.SiteFooter && SiteFooter.refresh) SiteFooter.refresh(project);
  }

  function showView(id) {
    ["view-landing", "view-hub", "view-location", "view-completion", "view-teacher"].forEach((v) => {
      const el = document.getElementById(v);
      if (el) el.classList.toggle("hidden", v !== id);
    });
    const header = document.getElementById("appHeader");
    if (header) {
      header.classList.toggle("hidden", id === "view-landing");
    }
    const maxW = document.getElementById("mainContainer");
    if (maxW) {
      maxW.classList.toggle("teacher-wide", id === "view-teacher");
    }
    const headerInner = document.getElementById("headerInner");
    if (headerInner) {
      headerInner.classList.toggle("max-w-md", id !== "view-teacher");
      headerInner.classList.toggle("max-w-4xl", id === "view-teacher");
    }
  }

  async function showHub(opts) {
    opts = opts || {};
    currentLocId = null;
    showView("view-hub");
    document.getElementById("hubTitle").textContent = project.title || "考察任務";
    document.getElementById("hubSubtitle").textContent = project.subtitle || "";
    if (!opts.skipRefresh) {
      await refreshTeamSessionFromServer({ force: true, refreshUi: false });
    }
    renderHubCards();
    updateHeader();
    window.location.hash = "#/hub";
  }

  function renderHubCards() {
    const container = document.getElementById("hubCards");
    container.innerHTML = "";
    const r = radius();
    locations.forEach((loc) => {
      const needsGps = locationRequiresGps(loc);
      let distance = null;
      let unlocked = testMode || !needsGps;
      if (needsGps && currentCoords) {
        distance = GPS.calculateDistance(
          currentCoords.lat,
          currentCoords.lng,
          loc.lat,
          loc.lng
        );
        if (distance <= r) unlocked = true;
      } else if (needsGps && currentCoords == null) {
        // waiting GPS
      } else if (!needsGps && currentCoords) {
        distance = GPS.calculateDistance(
          currentCoords.lat,
          currentCoords.lng,
          loc.lat,
          loc.lng
        );
      }
      const done = isLocComplete(loc.id);
      const card = document.createElement("button");
      card.type = "button";
      card.className =
        "w-full text-left p-4 rounded-2xl border shadow-sm transition " +
        (done
          ? "bg-emerald-50 border-emerald-300"
          : unlocked
            ? "bg-white border-emerald-200 hover:border-emerald-400"
            : "bg-slate-50 border-slate-200");
      // Rainbow / 已到達 framing only for GPS-required locations that are in range
      if (needsGps && unlocked && !done && !testMode) {
        card.classList.add("rainbow-arrived");
      }
      let distText = "";
      if (!needsGps) {
        distText = "不須到達";
      } else if (distance != null) {
        distText = "距離約 " + Math.round(distance) + " 米";
      } else {
        distText = "等待 GPS…";
      }
      let statusBadge = "";
      if (done) {
        statusBadge = '<span class="badge-done">已完成</span>';
      } else if (!needsGps) {
        // no 未到達 / 可進入 GPS framing — always enterable
        statusBadge = "";
      } else if (unlocked) {
        statusBadge = '<span class="badge-open">可進入</span>';
      } else {
        statusBadge = '<span class="badge-lock">未到達</span>';
      }
      card.innerHTML =
        '<div class="flex justify-between items-start gap-2">' +
        '<div><h3 class="font-bold text-slate-800 text-base">' +
        escapeHtml(loc.name) +
        "</h3>" +
        '<p class="text-xs text-slate-500 mt-1 line-clamp-2">' +
        escapeHtml(loc.description || "") +
        "</p></div>" +
        statusBadge +
        "</div>" +
        '<div class="mt-3 flex justify-between items-center text-xs text-slate-500">' +
        "<span>" +
        (needsGps
          ? '<i class="fa-solid fa-location-dot mr-1"></i>'
          : '<i class="fa-solid fa-unlock mr-1 text-slate-400"></i>') +
        distText +
        (needsGps && unlocked && !done ? " · 範圍內" : "") +
        "</span>" +
        '<span class="font-medium text-emerald-700"><i class="fa-solid fa-arrow-right"></i> 進入</span>' +
        "</div>";
      card.onclick = () => openLocation(loc.id);
      container.appendChild(card);
    });
  }

  async function openLocation(locId, opts) {
    opts = opts || {};
    const loc = locations.find((l) => l.id === locId);
    if (!loc) return;
    if (!opts.skipRefresh) {
      await refreshTeamSessionFromServer({ force: true, refreshUi: false });
    }
    currentLocId = locId;
    showView("view-location");
    window.location.hash = "#/loc/" + locId;

    document.getElementById("locPageTitle").textContent = loc.name;
    document.getElementById("locPageDesc").textContent = loc.description || "";
    const imgWrap = document.getElementById("locPageImages");
    imgWrap.innerHTML = "";
    (loc.images || []).forEach((url) => {
      if (!url) return;
      const img = document.createElement("img");
      img.src = url;
      img.alt = loc.name;
      img.className = "rounded-xl w-full object-cover max-h-48 border border-slate-200 img-zoomable";
      img.loading = "lazy";
      img.onclick = function () {
        openLightbox(url, loc.name);
      };
      imgWrap.appendChild(img);
    });
    if (!(loc.images || []).length) {
      const needsGpsHint = locationRequiresGps(loc);
      imgWrap.innerHTML =
        '<div class="bg-slate-100 rounded-xl p-6 text-center text-slate-500 text-sm">' +
        '<i class="fa-solid fa-map-location-dot text-2xl text-emerald-500 mb-2"></i>' +
        "<p>" +
        (needsGpsHint
          ? "請前往此地區；到達 GPS 範圍後會顯示問題。"
          : "此地區不須到達 GPS，可直接回答下方問題。") +
        "</p></div>";
    }

    const r = radius();
    const needsGps = locationRequiresGps(loc);
    let unlocked = testMode || !needsGps;
    let distance = null;
    if (currentCoords) {
      distance = GPS.calculateDistance(
        currentCoords.lat,
        currentCoords.lng,
        loc.lat,
        loc.lng
      );
      if (needsGps && distance <= r) unlocked = true;
    }

    const lockBanner = document.getElementById("locLockBanner");
    const qSection = document.getElementById("locQuestionsSection");
    const infoCard = document.getElementById("locInfoCard");
    const done = isLocComplete(loc.id);
    const showQuestions = unlocked || !needsGps;

    if (!needsGps || unlocked) {
      lockBanner.classList.add("hidden");
    } else {
      lockBanner.classList.remove("hidden");
      const lockTitle = lockBanner.querySelector("p.font-bold");
      if (lockTitle) {
        lockTitle.innerHTML =
          '<i class="fa-solid fa-lock"></i> 問題尚未解鎖';
      }
      let lockMsg =
        distance != null
          ? "尚未到達（約 " +
            Math.round(distance) +
            " 米，需在 " +
            r +
            " 米內）。"
          : "正在等待 GPS…（需在 " + r + " 米範圍內）。";
      lockMsg += "請參考上方指引前往；到達後才會顯示問題。";
      document.getElementById("locLockMsg").textContent = lockMsg;
    }

    if (showQuestions) {
      qSection.classList.remove("hidden");
      const showRainbow = needsGps && unlocked && !done && !testMode;
      if (infoCard) {
        infoCard.classList.toggle("rainbow-arrived", showRainbow);
      }
      qSection.classList.toggle("rainbow-arrived", showRainbow);
      renderQuestions(loc, isSubmissionLocked(loc.id), true);
    } else {
      qSection.classList.add("hidden");
      if (infoCard) infoCard.classList.remove("rainbow-arrived");
      qSection.classList.remove("rainbow-arrived");
    }
    updateHeader();
  }
