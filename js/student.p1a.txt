/**
 * 學生／隊伍端：登入後 Hub、地區內容頁、GPS 解鎖、提交
 */
window.StudentUI = (function () {
  let project = null;
  let locations = [];
  let session = null;
  let teamSession = null;
  let currentCoords = null;
  let lastReportedCoords = null;
  let lastGpsReportAt = 0;
  let gpsWatchStarted = false;
  let gpsAutoTimer = null;
  let sessionRefreshTimer = null;
  let sessionRefreshBound = false;
  let lastSessionFetchAt = 0;
  let testMode = false;
  let currentLocId = null;
  let draftAnswers = {}; // locationId -> { qId: value } // unsubmitted only

  const DRAFT_PREFIX = "gpsgame2_draft_";

  function draftKey() {
    return DRAFT_PREFIX + (session ? session.projectId + "_" + session.teamId : "");
  }

  function loadDrafts() {
    try {
      draftAnswers = JSON.parse(localStorage.getItem(draftKey()) || "{}");
    } catch (e) {
      draftAnswers = {};
    }
  }

  function saveDrafts() {
    try {
      localStorage.setItem(draftKey(), JSON.stringify(draftAnswers));
    } catch (e) {
      /* ignore quota */
    }
  }

  function clearAllDrafts() {
    draftAnswers = {};
    try {
      localStorage.removeItem(draftKey());
    } catch (e) {}
  }

  function serverHasAnyLocationAnswers(sess) {
    const map = (sess && sess.locationAnswers) || {};
    const keys = Object.keys(map);
    for (let i = 0; i < keys.length; i++) {
      const ans = map[keys[i]];
      if (ans && typeof ans === "object" && Object.keys(ans).length) return true;
    }
    return false;
  }

  /**
   * Firebase is source of truth for progress. localStorage drafts are only for
   * unsubmitted answers. When teacher clears project data (empty completions +
   * no locationAnswers), wipe this team's drafts so phones stop showing old work.
   */
  function reconcileDraftsWithServer(prevSession, nextSession) {
    const nextIds = (nextSession && nextSession.completedLocationIds) || [];
    const prevIds = (prevSession && prevSession.completedLocationIds) || [];
    if (nextIds.length === 0 && !serverHasAnyLocationAnswers(nextSession)) {
      clearAllDrafts();
      return;
    }
    const nextSet = new Set(nextIds);
    const unlocked = (nextSession && nextSession.unlockedEdits) || {};
    let changed = false;
    // Drop drafts for locations no longer completed on server (cleared)
    prevIds.forEach(function (id) {
      if (!nextSet.has(id) && draftAnswers[id]) {
        delete draftAnswers[id];
        changed = true;
      }
    });
    // Drop drafts for server-completed locations that are locked (answers live on server)
    nextIds.forEach(function (id) {
      if (!unlocked[id] && draftAnswers[id]) {
        delete draftAnswers[id];
        changed = true;
      }
    });
    // Also drop drafts whose locationAnswers were cleared on server for that loc
    const locAns = (nextSession && nextSession.locationAnswers) || {};
    Object.keys(draftAnswers).forEach(function (locId) {
      if (nextSet.has(locId)) return;
      const serverAns = locAns[locId];
      if (!serverAns || !Object.keys(serverAns).length) {
        // keep unsubmitted drafts for incomplete locations
        return;
      }
    });
    if (changed) saveDrafts();
  }

  async function refreshTeamSessionFromServer(opts) {
    opts = opts || {};
    if (!session || !project) return teamSession;
    const now = Date.now();
    const minGap = opts.force ? 0 : 4000;
    if (!opts.force && lastSessionFetchAt && now - lastSessionFetchAt < minGap) {
      return teamSession;
    }
    const prev = teamSession;
    try {
      const next = await DataStore.getTeamSession(project.id, session.teamId);
      lastSessionFetchAt = Date.now();
      teamSession = next || {
        lastLoginAt: null,
        completedLocationIds: [],
        locationAnswers: {},
        unlockedEdits: {}
      };
      reconcileDraftsWithServer(prev, teamSession);
      if (opts.refreshUi !== false) {
        applySessionUiRefresh();
      }
    } catch (e) {
      console.warn("refreshTeamSessionFromServer failed", e && e.message);
    }
    return teamSession;
  }

  function applySessionUiRefresh() {
    updateHeader();
    const hub = document.getElementById("view-hub");
    const locView = document.getElementById("view-location");
    const comp = document.getElementById("view-completion");
    if (comp && !comp.classList.contains("hidden") && !isAllComplete()) {
      showHub({ skipRefresh: true });
      return;
    }
    if (currentLocId && locView && !locView.classList.contains("hidden")) {
      openLocation(currentLocId, { skipRefresh: true });
      return;
    }
    if (hub && !hub.classList.contains("hidden")) {
      renderHubCards();
    }
  }

  function bindSessionRefreshListeners() {
    if (sessionRefreshBound) return;
    sessionRefreshBound = true;
    document.addEventListener("visibilitychange", function () {
      if (document.visibilityState !== "visible") return;
      if (!session || !project) return;
      refreshTeamSessionFromServer({ force: true });
    });
    window.addEventListener("focus", function () {
      if (!session || !project) return;
      refreshTeamSessionFromServer({ force: true });
    });
  }

  function startSessionRefreshInterval() {
    if (sessionRefreshTimer) return;
    const INTERVAL = 45000;
    sessionRefreshTimer = setInterval(function () {
      if (!session || !project) return;
      if (document.visibilityState === "hidden") return;
      refreshTeamSessionFromServer({ force: true });
    }, INTERVAL);
  }

  function applyFontSize() {
    const size = (project && project.fontSize) || "base";
    document.documentElement.style.setProperty(
      "--app-font-scale",
      size === "lg" ? "1.125" : size === "sm" ? "0.9" : "1"
    );
    document.body.classList.toggle("text-lg-mode", size === "lg");
    document.body.classList.toggle("text-sm-mode", size === "sm");
  }

  function radius() {
    return (project && project.gpsRadiusMeters) || 50;
  }

  function completedSet() {
    return new Set((teamSession && teamSession.completedLocationIds) || []);
  }

  function isEditUnlocked(locId) {
    const map = (teamSession && teamSession.unlockedEdits) || {};
    return !!map[locId];
  }

  function isSubmissionLocked(locId) {
    if (!completedSet().has(locId)) return false;
    return !isEditUnlocked(locId);
  }

  function answersForLocation(locId) {
    const fromSession =
      teamSession &&
      teamSession.locationAnswers &&
      teamSession.locationAnswers[locId];
    const server =
      fromSession && typeof fromSession === "object"
        ? Object.assign({}, fromSession)
        : null;
    const draft = draftAnswers[locId] || {};
    const completed = completedSet().has(locId);
    const canEditDraft = !completed || isEditUnlocked(locId);
    if (server && Object.keys(server).length) {
      if (!canEditDraft) return server;
      // Prefer server submission; draft only fills unanswered gaps
      Object.keys(draft).forEach(function (qid) {
        const sv = server[qid];
        if (sv == null || String(sv).trim() === "") {
          if (draft[qid] != null && String(draft[qid]).trim() !== "") {
            server[qid] = draft[qid];
          }
        }
      });
      return server;
    }
    if (canEditDraft) return Object.assign({}, draft);
    return {};
  }

  /** Progress squares / hub cards: Firebase completedLocationIds only (never localStorage). */
  function isLocComplete(locId) {
    return completedSet().has(locId);
  }

  /** Default true（須到達）when missing — backward compatible. */
  function questionRequiresGps(q) {
    return !q || q.requireGps !== false;
  }

  function locationHasOptionalQuestions(loc) {
    return (loc.questions || []).some(function (q) {
      return !questionRequiresGps(q);
    });
  }

  function isLocationGpsUnlocked(loc) {
    if (testMode) return true;
    if (!currentCoords || !loc) return false;
    const r = radius();
    const distance = GPS.calculateDistance(
      currentCoords.lat,
      currentCoords.lng,
      loc.lat,
      loc.lng
    );
    return distance <= r;
  }

  function countQuestions() {
    let n = 0;
    locations.forEach(function (loc) {
      n += (loc.questions || []).length;
    });
    return n;
  }

  function isAllComplete() {
    if (!locations.length) return false;
    const done = completedSet();
    // Every location that has questions must be submitted/completed
    let anyQ = false;
    for (let i = 0; i < locations.length; i++) {
      const loc = locations[i];
      const qs = loc.questions || [];
      if (!qs.length) continue;
      anyQ = true;
      if (!done.has(loc.id)) return false;
    }
    // If no questions at all, not "complete" in celebration sense
    return anyQ;
  }

  function openLightbox(src, alt) {
    const overlay = document.getElementById("imgLightbox");
    const img = document.getElementById("imgLightboxImg");
    if (!overlay || !img || !src) return;
    img.src = src;
    img.alt = alt || "放大圖片";
    overlay.classList.remove("hidden");
  }

  function closeLightbox() {
    const overlay = document.getElementById("imgLightbox");
    const img = document.getElementById("imgLightboxImg");
    if (overlay) overlay.classList.add("hidden");
    if (img) img.src = "";
  }

  function showCompletion() {
    currentLocId = null;
    showView("view-completion");
    const name = (session && session.teamName) || "隊伍";
    const p = project || {};
    const images = (p.completionImages || []).filter(Boolean);
    const imgWrap = document.getElementById("completionImages");
    const trophy = document.getElementById("completionTrophy");
    if (imgWrap) {
      imgWrap.innerHTML = "";
      images.forEach(function (url) {
        const img = document.createElement("img");
        img.src = url;
        img.alt = p.completionTitle || "完成圖片";
        img.className =
          "rounded-xl w-full object-cover max-h-56 border border-slate-200 img-zoomable";
        img.loading = "lazy";
        img.onclick = function () {
          openLightbox(url, p.completionTitle || "完成圖片");
        };
        imgWrap.appendChild(img);
      });
    }
    if (trophy) {
      // Keep trophy when no custom images; still show with images (title stays)
      trophy.classList.toggle("hidden", images.length > 0);
    }
    const titleEl = document.getElementById("completionTitle");
    if (titleEl) {
      titleEl.textContent =
        (p.completionTitle && String(p.completionTitle).trim()) || "恭喜完成！";
    }
    const congrats = document.getElementById("completionCongrats");
    if (congrats) {
      const rawMsg =
        (p.completionMessage && String(p.completionMessage).trim()) ||
        "「{teamName}」已完成所有考察問題，辛苦了！";
      congrats.textContent = rawMsg.split("{teamName}").join(name);
    }
    const summary = document.getElementById("completionSummary");
    const showStats = p.completionShowStats !== false;
    if (summary) {
      if (!showStats) {
        summary.classList.add("hidden");
        summary.innerHTML = "";
      } else {
        summary.classList.remove("hidden");
        const doneLocs = locations.filter(function (loc) {
          return isLocComplete(loc.id);
        }).length;
        const totalQ = countQuestions();
        let locLines = locations
          .map(function (loc, i) {
            const qs = (loc.questions || []).length;
            const ok = isLocComplete(loc.id);
            return (
              '<div class="flex justify-between gap-2 py-0.5 border-b border-emerald-100/80 last:border-0">' +
              "<span>" +
              escapeHtml(loc.name || "地區 " + (i + 1)) +
              '</span><span class="' +
              (ok ? "text-emerald-700 font-semibold" : "text-slate-400") +
              '">' +
              (ok ? "已完成" : "未完成") +
              (qs ? " · " + qs + " 題" : "") +
              "</span></div>"
            );
          })
          .join("");
        summary.innerHTML =
          '<p><span class="text-slate-500">隊伍：</span><strong>' +
          escapeHtml(name) +
          "</strong></p>" +
          '<p><span class="text-slate-500">地區：</span><strong>' +
          doneLocs +
          " / " +
          locations.length +
          "</strong></p>" +
          '<p><span class="text-slate-500">問題：</span><strong>' +
          totalQ +
          " 題</strong></p>" +
          '<div class="mt-2">' +
          locLines +
          "</div>";
      }
    }
    updateHeader();
    window.location.hash = "#/complete";
  }
