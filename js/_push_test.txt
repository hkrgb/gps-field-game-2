  function collectCurrentAnswers(loc) {
    const answers = Object.assign({}, drafts[loc.id] || {});
    (loc.questions || []).forEach((q) => {
      const el = document.getElementById("ans_" + q.id);
      if (!el) return;
      answers[q.id] = el.value;
    });
    return answers;
  }

  async function submitCurrentLocation() {
    if (!currentLocId) return;
    const loc = locations.find((l) => l.id === currentLocId);
    if (!loc) return;
    if (isLocationLocked(loc.id)) {
      alert("此地區已提交並鎖定，無法再修改。請老師在監察頁解鎖後再試。");
      return;
    }
    const answers = collectCurrentAnswers(loc);
    const missing = (loc.questions || []).filter((q) => !String(answers[q.id] || "").trim());
    if (missing.length) {
      if (!confirm("尚有 " + missing.length + " 題未填，仍要提交？")) return;
    }
    StudentUI.showLoading(true);
    try {
      await DataStore.submitAnswers(project.id, session.teamId, loc.id, answers, currentCoords);
      teamSession = await DataStore.getTeamSession(project.id, session.teamId);
      delete drafts[loc.id];
      saveDrafts();
      updateHeader();
      alert("已提交！");
      if (isAllComplete()) showCompletion();
      else showHub();
    } catch (e) {
      alert(e.message || "提交失敗");
    } finally {
      StudentUI.showLoading(false);
    }
  }
